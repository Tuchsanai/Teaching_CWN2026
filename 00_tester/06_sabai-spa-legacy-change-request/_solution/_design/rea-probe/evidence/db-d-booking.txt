Check constraints:
    "booking_booking_type_check" CHECK (booking_type::text = ANY (ARRAY['single'::character varying, 'couple'::character varying, 'group'::character varying, 'package'::character varying, 'course'::character varying]::text[]))
    "booking_check" CHECK (end_at > start_at)
    "booking_check1" CHECK (customer_id IS NOT NULL OR guest_name IS NOT NULL)
    "booking_check2" CHECK (booking_type::text <> 'couple'::text OR party_size = 2 AND room_id IS NOT NULL)
    "booking_check3" CHECK (booking_type::text <> 'group'::text OR party_size >= 3 AND party_size <= 6)
    "booking_check4" CHECK (booking_type::text <> 'single'::text OR party_size = 1)
    "booking_check5" CHECK (booking_type::text <> 'course'::text OR party_size = 1 AND course_purchase_id IS NOT NULL)
    "booking_check6" CHECK ((giftcard_paid + other_paid) = total_amount)
    "booking_discount_amount_check" CHECK (discount_amount >= 0::numeric)
    "booking_discount_kind_check" CHECK (discount_kind::text = ANY (ARRAY['none'::character varying, 'group'::character varying, 'tier'::character varying, 'birthday'::character varying]::text[]))
    "booking_discount_pct_check" CHECK (discount_pct >= 0::numeric AND discount_pct <= 100::numeric)
    "booking_distance_km_check" CHECK (distance_km IS NULL OR distance_km >= 0::numeric AND distance_km <= 15.0)
    "booking_giftcard_paid_check" CHECK (giftcard_paid >= 0::numeric)
    "booking_gross_amount_check" CHECK (gross_amount >= 0::numeric)
    "booking_other_paid_check" CHECK (other_paid >= 0::numeric)
    "booking_party_size_check" CHECK (party_size >= 1 AND party_size <= 6)
    "booking_points_discount_check" CHECK (points_discount >= 0::numeric)
    "booking_points_earned_check" CHECK (points_earned >= 0)
    "booking_points_redeemed_check" CHECK (points_redeemed >= 0 AND (points_redeemed % 100) = 0)
    "booking_status_check" CHECK (status::text = ANY (ARRAY['confirmed'::character varying, 'cancelled'::character varying, 'completed'::character varying]::text[]))
    "booking_total_amount_check" CHECK (total_amount >= 0::numeric)
    "booking_transport_fee_check" CHECK (transport_fee >= 0::numeric)
    "booking_vehicles_check" CHECK (vehicles >= 0)
