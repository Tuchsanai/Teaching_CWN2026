## 3. ค่าที่คาดหวังใหม่ (SQL ตรงกับฐาน หลัง reset · `now` = 2026-10-10 10:00)

SQL ทั้งหมด (`tools`/scratch ของ runner: `expected.sql` · รันด้วย `docker cp` + `psql -v ON_ERROR_STOP=1 -f` · exit 0) · ช่วง = `(b.start_at AT TIME ZONE 'Asia/Bangkok')::date BETWEEN from AND to`

```sql
-- rb = นัดในช่วง (วันนัดเวลาไทย รวมหัวท้าย)
CREATE TEMP VIEW r AS SELECT * FROM (VALUES ('A_sep','2026-09-01'::date,'2026-09-30'::date),
  ('B_oct','2026-10-01','2026-10-31'),('C_6_9oct','2026-10-06','2026-10-09'),('D_aug_oct92','2026-08-01','2026-10-31')) v(rng,d_from,d_to);
CREATE TEMP VIEW rb AS SELECT r.rng, b.* FROM r JOIN booking b ON (b.start_at AT TIME ZONE 'Asia/Bangkok')::date BETWEEN r.d_from AND r.d_to;
-- q_counts
SELECT r.rng, count(rb.id) FILTER (WHERE rb.status='confirmed'), count(rb.id) FILTER (WHERE rb.status='completed'),
       count(rb.id) FILTER (WHERE rb.status='cancelled'), count(rb.id) AS csv_rows FROM r LEFT JOIN rb ON rb.rng=r.rng GROUP BY r.rng;
-- q_money (completed เท่านั้น)
SELECT r.rng, sum(total_amount) revenue, sum(giftcard_paid), sum(other_paid), sum(transport_fee),
       sum(discount_amount) FILTER (WHERE discount_kind='group'|'tier'|'birthday') ..., sum(points_earned),
       count(*) FILTER (WHERE course_purchase_id IS NOT NULL) course_sessions
  FROM r LEFT JOIN rb ON rb.rng=r.rng AND rb.status='completed' GROUP BY r.rng;
-- q_points_txn (ไขว้) : sum(point_txn.points) reason='earn' ของนัดในช่วง
-- q_course_txn (ไขว้) : count completed booking_type='course'
-- q_pending : sum(total_amount) WHERE status='confirmed'
-- q_therapist : therapist CROSS JOIN · LEFT JOIN booking_slot → rb completed · count(DISTINCT rb.id)
-- q_csv_sep : แถว ก.ย. เรียง start_at, id · q_tz : count วัน UTC ≠ วันไทย · q_days : to − from + 1
```
