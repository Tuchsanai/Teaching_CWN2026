def o_cashback(amount, tier, birthday):        # คำตอบที่ถูก (Oracle)
    if amount < 10000:                         #   ข้อกำหนดที่ระบุชัด: ต่ำกว่า 100 บาท ไม่ได้เงินคืน
        return 0
    x = amount * RATE[tier]                    #   ข้อกำหนดที่ระบุชัด: อัตราตามระดับสมาชิก
    if birthday and tier in ("gold", "platinum"):
        x *= 2                                 #   ระบุชัด: เดือนเกิด 2 เท่า + นโยบาย C1: เฉพาะ gold platinum
    return min(20000, int(F(x) // 1))          #   ระบุชัด: สูงสุด 200 บาท + นโยบาย C2: ปัดลง

RATES_BP = {"silver": 150, "gold": 250, "platinum": 400}  # basis points (1% = 100)
MIN_AMOUNT = 10_000   # 100 baht in satang
MAX_CASHBACK = 20_000  # 200 baht in satang


def cashback(amount, tier, birthday):
    """Return cashback in whole satang (rounded down)."""
    if amount < MIN_AMOUNT:
        return 0
    mult = 2 if birthday else 1
    result = amount * RATES_BP[tier] * mult // 10_000  # integer math, no float error
    return min(result, MAX_CASHBACK)

import json
import sys

_RATES = {"silver": 15, "gold": 25, "platinum": 40}  # ส่วน 1000
_DEN = 1000
_MIN_AMOUNT = 100 * 100   # 100 บาท (สตางค์)
_CAP = 200 * 100          # 200 บาท (สตางค์)


def cashback(amount, tier, birthday):
    if type(amount) is not int or amount < 0:
        raise ValueError("amount must be a non-negative int")
    if type(tier) is not str or tier not in _RATES:
        raise ValueError("invalid tier")
    if type(birthday) is not bool:
        raise ValueError("birthday must be bool")

    if amount < _MIN_AMOUNT:
        return 0
    mult = 2 if birthday and tier in ("gold", "platinum") else 1
    num = amount * _RATES[tier] * mult  # ตัวเศษ ตัวส่วน = 1000 (ค่าแม่นตรง)
    num = min(num, _CAP * _DEN)
    return num // _DEN  # เงินคืนปัดลงเสมอ

สกิลรุ่น v3 · ส่วน CLI · เปิดไฟล์ solve.py

def _main():
    try:
        data = json.loads(sys.stdin.read())
        if not isinstance(data, dict):
            raise ValueError("input must be a JSON object")
        for k in ("amount", "tier", "birthday"):
            if k not in data:
                raise ValueError("missing key: " + k)
        result = cashback(data["amount"], data["tier"], data["birthday"])
    except (ValueError, TypeError) as e:
        print(json.dumps({"ok": False, "error": str(e)}))
        return 2
    print(json.dumps({"ok": True, "result": result}))
    return 0


if __name__ == "__main__":
    sys.exit(_main())
