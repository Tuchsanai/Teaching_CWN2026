def shipping_fee(weight_g, distance_km, express):
    """ค่าส่งเป็นจำนวนเต็มสตางค์ (ปัดครึ่งขึ้นเมื่อมีเศษสตางค์)"""

    if distance_km > 100:
        fee *= Fraction(6, 5)
    if express:
        fee *= Fraction(3, 2)
    return int(floor(fee + Fraction(1, 2)))
