28	
29	def half_up(x):
30	    """ปัดครึ่งขึ้นเป็น int, x ไม่ติดลบ (int หรือ Fraction)"""
31	    return int((Fraction(x) * 2 + 1) // 2)
32	
33	def floor_int(x):
34	    """ปัดลงเป็น int, x ไม่ติดลบ"""
35	    return int(Fraction(x) // 1)
36	
37	def percent_of(amount, pct):
38	    """amount * pct% โดยยังไม่ปัด คืนเป็น Fraction"""
39	    return Fraction(amount) * Fraction(pct, 100)
40	```

19	```python
20	def require_int(value, name, low=None, high=None):
21	    if type(value) is not int:          # ปฏิเสธ bool, float, str, None
22	        raise ValueError(f"{name} ต้องเป็นจำนวนเต็ม")
23	    if low is not None and value < low:
24	        raise ValueError(f"{name} ต้องไม่น้อยกว่า {low}")
25	    if high is not None and value > high:
26	        raise ValueError(f"{name} ต้องไม่เกิน {high}")
27	    return value
28	
29	def require_bool(value, name):
30	    if type(value) is not bool:
31	        raise ValueError(f"{name} ต้องเป็น True หรือ False")
32	    return value
33	
34	def require_choice(value, name, choices):
35	    if type(value) is not str or value not in choices:   # ไม่ normalize
36	        raise ValueError(f"{name} ต้องเป็นหนึ่งใน {sorted(choices)}")
37	    return value
38	
39	def require_list_of_dicts(value, name):
40	    if type(value) is not list:
41	        raise ValueError(f"{name} ต้องเป็น list")
42	    for entry in value:
43	        if type(entry) is not dict:
